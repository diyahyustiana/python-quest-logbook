Deploy > Kelola deployment > ikon pensil > Versi: "Versi baru" > Deploy. const PIN_AWAL = 'diyah1981'; // dipakai sampai PIN diganti lewat "Lupa PIN?" const GURU_EMAIL = 'diyah.smkn1kediri@gmail.com'; // email penerima kode reset PIN (periksa: ganti jika bukan email yang dipakai) const FIELDS = { log: ['ts','kelas','absen','nama','tanggal','pertemuan','tahap','target','dikerjakan','commit','error','solusi','progres','status','kendala','rencana'], ai: ['ts','kelas','absen','nama','tanggal','masalah','upaya','prompt','saran','diuji','porsi','penjelasan','refleksi'] }; const SHEET_NAME = { log: 'LogSesi', ai: 'AICheckpoint' }; const RESET_JEDA_MS = 2 * 60 * 1000; const RESET_BERLAKU_MS = 15 * 60 * 1000; // ---------- utilitas ---------- const props = () => PropertiesService.getScriptProperties(); const out = o => ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); const safe = v => (typeof v === 'string' && /^[=+\-@]/.test(v)) ? "'" + v : v; // cegah rumus const norm = s => String(s).toLowerCase().replace(/\s+/g, ' ').trim(); const hash = s => Utilities.base64Encode(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, 'pqlog:' + s)); const maskEmail = e => { const p = String(e).split('@'); return p[0].slice(0, 2) + '***@' + (p[1] || ''); }; function emailGuru() { let e = String(GURU_EMAIL || '').trim(); if (!e) { try { e = Session.getEffectiveUser().getEmail(); } catch (x) {} } return e; } function checkPin(pin) { const p = String(pin || '').trim(), h = props().getProperty('PIN_HASH'); return h ? hash(p) === h : p === String(PIN_AWAL).trim(); } function sheetFor(type) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME[type]); if (!sh) { sh = ss.insertSheet(SHEET_NAME[type]); sh.getRange(1, 1, sh.getMaxRows(), FIELDS[type].length).setNumberFormat('@'); // teks biasa: tanggal/angka tidak diubah otomatis oleh Sheets sh.appendRow(FIELDS[type]); sh.setFrozenRows(1); } return sh; } function readAll(type) { const tz = SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone(); const v = sheetFor(type).getDataRange().getValues(); const h = v.shift().map(x => String(x).trim()); return v.filter(row => row.some(c => c !== '')).map(row => { const o = {}; h.forEach((k, i) => { if (!k) return; let c = row[i]; if (c instanceof Date) c = (k === 'ts') ? c.toISOString() : Utilities.formatDate(c, tz, 'yyyy-MM-dd'); o[k] = (c === null || c === undefined) ? '' : c; }); return o; }); } // ---------- daftar siswa (sheet "data": Kelas, No Absen, Nama) ---------- function getRoster() { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheets().find(x => x.getName().toLowerCase().trim() === 'data'); if (!sh) return {}; const v = sh.getDataRange().getDisplayValues(); let hr = -1, ik = -1, ia = -1, inm = -1; for (let i = 0; i < Math.min(v.length, 15); i++) { const h = v[i].map(x => String(x).toLowerCase().trim()); const k = h.findIndex(x => x.includes('kelas')), n = h.findIndex(x => x.includes('nama')); if (k >= 0 && n >= 0) { hr = i; ik = k; inm = n; ia = h.findIndex(x => x.includes('absen') || x === 'no' || x === 'no.' || x.includes('nomor') || x.includes('urut')); break; } } if (hr < 0) return {}; const res = {}; let kelas = ''; v.slice(hr + 1).forEach(r => { const k = String(r[ik]).trim(); if (k) kelas = k; const n = String(r[inm]).trim().replace(/\s+/g, ' '); if (!kelas || !n) return; (res[kelas] = res[kelas] || []).push({ absen: ia >= 0 ? r[ia] : '', nama: n }); }); Object.keys(res).forEach(k => { res[k].forEach((x, i) => { const a = parseInt(String(x.absen).replace(/\D/g, ''), 10); x.absen = isNaN(a) ? i + 1 : a; }); res[k].sort((a, b) => a.absen - b.absen); }); return res; } // ---------- reset PIN ---------- function mailError(err) { return /izin|permission|authori|send_mail|scope/i.test(String(err)) ? 'Izin kirim email belum diberikan. Guru: di editor Apps Script jalankan fungsi "setup", setujui izinnya, lalu deploy versi baru.' : 'Email gagal dikirim: ' + String(err).slice(0, 140); } function resetRequest() { const to = emailGuru(); if (!to) return { ok: false, error: 'Email guru belum diketahui. Isi GURU_EMAIL di Code.gs, lalu deploy versi baru.' }; const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const P = props(), now = Date.now(), last = Number(P.getProperty('RESET_LAST') || 0); if (now - last < RESET_JEDA_MS) return { ok: false, error: 'Kode baru saja dikirim. Periksa kotak masuk dan folder spam, atau tunggu ' + Math.ceil((RESET_JEDA_MS - (now - last)) / 1000) + ' detik untuk meminta lagi.' }; const code = Utilities.getUuid().replace(/-/g, '').slice(0, 8).toUpperCase(); try { // kirim email DULU; jika gagal, tidak ada yang disimpan dan tidak ada jeda MailApp.sendEmail({ to: to, subject: 'Kode reset PIN Dashboard Python Quest Logbook', body: 'Kode reset PIN Dashboard Guru Anda: ' + code + '\n\nKode berlaku 15 menit dan hanya bisa dipakai sekali.\n' + 'Masukkan kode ini di Dashboard Guru > Lupa PIN, lalu buat PIN baru.\n\n' + 'Jika Anda tidak merasa meminta reset, abaikan email ini. PIN lama Anda tetap berlaku.' }); } catch (err) { return { ok: false, error: mailError(err) }; } P.setProperties({ RESET_CODE: hash(code), RESET_EXP: String(now + RESET_BERLAKU_MS), RESET_TRY: '0', RESET_LAST: String(now) }); return { ok: true, email: maskEmail(to) }; } finally { lock.releaseLock(); } } function resetConfirm(code, newPin) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const P = props(), h = P.getProperty('RESET_CODE'), exp = Number(P.getProperty('RESET_EXP') || 0), tries = Number(P.getProperty('RESET_TRY') || 0); const clear = () => ['RESET_CODE', 'RESET_EXP', 'RESET_TRY'].forEach(k => P.deleteProperty(k)); if (!h || Date.now() > exp) return { ok: false, error: 'Kode belum diminta atau sudah kedaluwarsa. Kirim kode baru.' }; if (tries >= 5) { clear(); return { ok: false, error: 'Terlalu banyak percobaan. Kirim kode baru.' }; } if (hash(code) !== h) { P.setProperty('RESET_TRY', String(tries + 1)); return { ok: false, error: 'Kode salah.' }; } const pin = String(newPin || '').trim(); if (pin.length < 6) return { ok: false, error: 'PIN baru minimal 6 karakter.' }; P.setProperty('PIN_HASH', hash(pin)); clear(); try { MailApp.sendEmail({ to: emailGuru(), subject: 'PIN Dashboard Python Quest Logbook telah diganti', body: 'PIN Dashboard Guru diganti pada ' + new Date().toLocaleString('id-ID') + '. Jika bukan Anda, segera minta reset ulang.' }); } catch (x) {} return { ok: true }; } finally { lock.releaseLock(); } } // ---------- pintu masuk web ---------- function doPost(e) { try { const d = JSON.parse(e.postData.contents); if (d.action === 'all') { if (!checkPin(d.pin)) return out({ ok: false, error: 'PIN salah.' }); return out({ ok: true, log: readAll('log'), ai: readAll('ai') }); } if (d.action === 'resetRequest') return out(resetRequest()); if (d.action === 'resetConfirm') return out(resetConfirm(String(d.code || '').trim().toUpperCase(), String(d.pin || ''))); if (!FIELDS[d.type]) return out({ ok: false, error: 'jenis tidak dikenal' }); const r = d.row || {}; if (!r.kelas || !r.absen || !r.nama) return out({ ok: false, error: 'identitas belum lengkap' }); const siswa = (getRoster()[r.kelas] || []).find(x => String(x.absen) === String(r.absen) && norm(x.nama) === norm(r.nama)); if (!siswa) return out({ ok: false, error: 'nama tidak cocok dengan daftar kelas' }); const lock = LockService.getScriptLock(); lock.waitLock(15000); try { const cache = CacheService.getScriptCache(), idKey = r.id ? 'sub_' + String(r.id).slice(0, 60) : null; if (idKey && cache.get(idKey)) return out({ ok: true, dup: true }); // kiriman ulang (jaringan putus) tidak digandakan sheetFor(d.type).appendRow(FIELDS[d.type].map(k => safe(r[k] ?? ''))); if (idKey) cache.put(idKey, '1', 21600); } finally { lock.releaseLock(); } return out({ ok: true }); } catch (err) { return out({ ok: false, error: String(err).slice(0, 160) }); } } function doGet(e) { try { const p = e.parameter || {}; if (p.action === 'roster') return out({ ok: true, roster: getRoster() }); if (p.action === 'mine') { const mine = x => x.kelas === p.kelas && String(x.absen) === String(p.absen) && norm(x.nama) === norm(p.nama || ''); return out({ ok: true, log: readAll('log').filter(mine), ai: readAll('ai').filter(mine) }); } if (p.action === 'ping') { // pemeriksaan kesehatan untuk guru let mailOk = false; try { MailApp.getRemainingDailyQuota(); mailOk = true; } catch (x) {} const r = getRoster(), to = emailGuru(); return out({ ok: true, versi: 3, kelas: Object.keys(r).length, siswa: Object.keys(r).reduce((a, k) => a + r[k].length, 0), email: to ? maskEmail(to) : '', emailOk: mailOk, pinTersimpan: !!props().getProperty('PIN_HASH') }); } return out({ ok: false, error: 'aksi tidak dikenal' }); } catch (err) { return out({ ok: false, error: String(err).slice(0, 160) }); } } // ---------- dijalankan manual dari editor ---------- // Jalankan SEKALI setelah menempel kode: memberi semua izin dan mengirim email uji. function setup() { const r = getRoster(), n = Object.keys(r).reduce((a, k) => a + r[k].length, 0); ['log', 'ai'].forEach(t => { const sh = sheetFor(t); sh.getRange(1, 1, sh.getMaxRows(), FIELDS[t].length).setNumberFormat('@'); }); const to = emailGuru(); if (!to) throw new Error('Isi GURU_EMAIL di bagian atas kode, simpan, lalu jalankan setup lagi.'); MailApp.sendEmail({ to: to, subject: 'Uji email Python Quest Logbook', body: 'Jika email ini diterima, pengiriman kode reset PIN siap dipakai.\nSiswa terbaca dari sheet "data": ' + n + ' di ' + Object.keys(r).length + ' kelas.' }); CacheService.getScriptCache().put('uji', '1', 10); Logger.log('Siap. Siswa terbaca: ' + n + '. Email uji dikirim ke ' + to + ' (cek juga folder spam).'); } // Cadangan jika email guru tidak bisa diakses: ubah PIN_BARU, lalu jalankan fungsi ini dari editor. function aturPinManual() { const PIN_BARU = 'ISI-PIN-BARU-DI-SINI'; if (PIN_BARU === 'ISI-PIN-BARU-DI-SINI' || PIN_BARU.length < 6) throw new Error('Ubah PIN_BARU di dalam fungsi ini (minimal 6 karakter), lalu jalankan lagi.'); props().setProperty('PIN_HASH', hash(PIN_BARU)); Logger.log('PIN Dashboard Guru sudah diganti.'); }
